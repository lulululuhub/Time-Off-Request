// ── ACCRUAL MATH ─────────────────────────────────────────────────
function yrs(sd){
  var s=new Date(sd),n=new Date();
  var y=n.getFullYear()-s.getFullYear();
  if(n<new Date(n.getFullYear(),s.getMonth(),s.getDate()))y--;
  return Math.max(0,y);
}
function allowance(e){return yrs(e.startDate)>=5?160:120;}
function cap(e){return allowance(e)*1.5;}
function rate(e){return allowance(e)/24;}
function conseq(e){return e.partTime?40:yrs(e.startDate)>=5?120:80;}

function getPeriodDates(sd){
  var hire=new Date(sd),today=new Date();
  var curYear=today.getFullYear();
  var yearStart=new Date(curYear,0,1);
  var effectiveStart=hire>yearStart?hire:yearStart;
  var dates=[];
  for(var mo=0;mo<12;mo++){
    var p15=new Date(curYear,mo,15);
    var pe=new Date(curYear,mo+1,0);
    var p15s=p15.toISOString().slice(0,10);
    var pes=pe.toISOString().slice(0,10);
    var effs=effectiveStart.toISOString().slice(0,10);
    var tods=today.toISOString().slice(0,10);
    if(p15s<tods&&p15s>=effs)dates.push(p15s);
    if(pes<tods&&pes>=effs)dates.push(pes);
  }
  return dates.sort();
}
function periods(sd){return getPeriodDates(sd).length;}

function bal(e,reqs,adjs){
  var DEDUCTIBLE=['Paid Time Off'];
  var ptoReqs=(reqs||[])
    .filter(function(r){return r.employeeEmail===e.email&&r.status==='Approved'&&DEDUCTIBLE.includes(r.type);})
    .map(function(r){return Object.assign({},r,{hours:Number(r.hours)});})
    .sort(function(a,b){return a.endDate>b.endDate?1:-1;});
  var empAdj=(adjs||[]).filter(function(a){return a.email===e.email;}).reduce(function(s,a){return s+Number(a.delta);},0);
  var balance=Math.min(Number(e.carryover||0)+empAdj,cap(e));
  var usedIdx=0;
  var periodDates=getPeriodDates(e.startDate);
  for(var i=0;i<periodDates.length;i++){
    var pd=periodDates[i];
    while(usedIdx<ptoReqs.length&&ptoReqs[usedIdx].endDate<pd){
      balance=Math.max(0,balance-ptoReqs[usedIdx].hours);usedIdx++;
    }
    balance=Math.min(balance+rate(e),cap(e));
  }
  while(usedIdx<ptoReqs.length){balance=Math.max(0,balance-ptoReqs[usedIdx].hours);usedIdx++;}
  return Math.max(0,balance);
}

function accrued(e){
  var b=Math.min(Number(e.carryover||0),cap(e));
  var pd=getPeriodDates(e.startDate);
  for(var i=0;i<pd.length;i++)b=Math.min(b+rate(e),cap(e));
  return b;
}

function pendHrs(e,reqs){
  return(reqs||[]).filter(function(r){return r.employeeEmail===e.email&&r.status==='Pending';}).reduce(function(s,r){return s+Number(r.hours);},0);
}

function nextAccrual(){
  var t=new Date(),d=t.getDate(),m=t.getMonth(),y=t.getFullYear();
  var last=new Date(y,m+1,0).getDate();
  var n;
  if(d<15)n=new Date(y,m,15);
  else if(d<last)n=new Date(y,m+1,0);
  else n=new Date(y,m+1,15);
  return n.toISOString().slice(0,10);
}

window.checkHols=function(){
  var s=document.getElementById('rstart')?document.getElementById('rstart').value:'';
  var e=document.getElementById('rend')?document.getElementById('rend').value:'';
  var el=document.getElementById('hol-warn');
  if(!el||!s||!e)return;
  var hits=HOLIDAYS_2026.filter(function(h){return h.date>=s&&h.date<=e;});
  el.innerHTML=hits.length>0?'<div class="alert info">Your date range includes '+hits.length+' company holiday(s): '+hits.map(function(h){return '<strong>'+h.name+'</strong>';}).join(', ')+'. Holidays do not count against your PTO balance.</div>':'';
};
window.dHint=function(v){
  var h=parseFloat(v)||0,el=document.getElementById('dhint');
  if(el&&h>0)el.textContent='approx '+(h/8).toFixed(2)+' days';
};

function calUI(showNames){
  var m=S.cal,y=m.getFullYear(),mo=m.getMonth();
  var first=new Date(y,mo,1).getDay(),days=new Date(y,mo+1,0).getDate();
  var td=new Date(),approved=S.reqs.filter(function(x){return x.status==='Approved';});
  var cells='';
  for(var i=0;i<first;i++)cells+='<div class="cal-cell other"></div>';
  for(var d=1;d<=days;d++){
    var ds=y+'-'+String(mo+1).padStart(2,'0')+'-'+String(d).padStart(2,'0');
    var evts=approved.filter(function(x){return x.startDate<=ds&&x.endDate>=ds;});
    var hol=getHoliday(ds);
    var isT=td.getFullYear()===y&&td.getMonth()===mo&&td.getDate()===d;
    cells+='<div class="cal-cell'+(isT?' today-cell':'')+(evts.length>0?' has-pto':'')+(hol?' has-holiday':'')+'">'
      +(isT?'<span class="cal-num-today">'+d+'</span>':'<span>'+d+'</span>')
      +(hol?'<div class="hol-dot" title="'+hol.name+'">'+hol.name.slice(0,10)+'</div>':'')
      +evts.map(function(x){
        var typeShort={'Paid Time Off':'PTO','Unpaid Time Off':'Unpaid','Bereavement':'Bereav.','Other':'Other'}[x.type]||x.type;
        var dotColor={'Paid Time Off':'#2d6a22','Unpaid Time Off':'#1a4d7c','Bereavement':'#7a7874','Other':'#b5651d'}[x.type]||'#2d6a22';

      }).join('')
      +'</div>';
  }
  var todayLocal=td.getFullYear()+'-'+String(td.getMonth()+1).padStart(2,'0')+'-'+String(td.getDate()).padStart(2,'0');

    +'<button class="btn-cal" onclick="S.cal=new Date();render()">Today</button>'
    +'<button class="btn-cal" onclick="S.cal=new Date('+y+','+(mo+1)+');render()">&#8594;</button>'
    +'</div>'
    +'<div class="cal-grid">'+['Sun','Mon','Tue','Wed','Thu','Fri','Sat'].map(function(x){return '<div class="cal-dow">'+x+'</div>';}).join('')+cells+'</div>'
    +'<div class="cal-legend"><span class="l-dot" style="background:var(--accent-light);border-color:#b8ddb5"></span>PTO <span class="l-dot" style="background:var(--holiday-light);border-color:#f5c6c6"></span>Holiday <span class="l-dot" style="background:var(--today-bg);border-color:var(--today-bd)"></span>Today</div>'
    +'</div>'
    +(thisMonthHols.length>0?'<div class="cal-wrap" style="margin-top:0"><div class="stitle">Holidays This Month</div>'

      +'</tbody></table></div>':'')

window.empCancel=function(id){
  if(!confirm('Cancel this request?'))return;
  S.reqs=S.reqs.map(function(x){return x.id===id?Object.assign({},x,{status:'Cancelled'}):x;});
  persist();toast('Request cancelled.','w');set({etab:'history'});
  if(USE_APPS_SCRIPT)gsCall('updateRequestStatus',id,'Cancelled').then(function(){return gsCall('getAllRequests');}).then(function(f){if(f)S.reqs=f;render();});
};
window.dHint=window.dHint||function(){};
window.doSubmit=function(){
  var emp=S.emps.find(function(e){return e.email===S.sess.email;});
  var type=document.getElementById('rtype').value;
  var start=document.getElementById('rstart').value;
  var end=document.getElementById('rend').value;
  var hours=parseFloat(document.getElementById('rhours').value)||0;
  var reason=document.getElementById('rreason').value;
  var errEl=document.getElementById('req-err');
  if(!start||!end){errEl.innerHTML='<div class="alert err">Please select dates.</div>';return;}
  if(new Date(start)>new Date(end)){errEl.innerHTML='<div class="alert err">End date must be on or after start date.</div>';return;}
  if(hours<4){errEl.innerHTML='<div class="alert err">Minimum request is 4 hours.</div>';return;}
  if(type==='Paid Time Off'){
    var b=bal(emp,S.reqs,S.adjs);
    if(hours>b){errEl.innerHTML='<div class="alert err">Insufficient balance. You have '+b.toFixed(2)+'h available.</div>';return;}
    if(hours>conseq(emp)){errEl.innerHTML='<div class="alert err">Exceeds consecutive limit of '+conseq(emp)+'h.</div>';return;}
  }
  var req={id:Date.now().toString(),employeeEmail:S.sess.email,employeeName:S.sess.name,type:type,startDate:start,endDate:end,hours:hours,reason:reason,status:'Pending',createdAt:new Date().toISOString(),managerEmail:emp.managerEmail};
  S.reqs=[].concat(S.reqs,[req]);
  persist();toast('Request submitted!','s');set({etab:'history'});
  if(USE_APPS_SCRIPT)gsCall('submitRequest',req).then(function(){return gsCall('getAllRequests');}).then(function(f){if(f)S.reqs=f;render();});
};
window.doStat=function(id,status){
  S.reqs=S.reqs.map(function(x){return x.id===id?Object.assign({},x,{status:status}):x;});
  persist();toast('Request '+status.toLowerCase()+'.', status==='Approved'?'s':'w');render();
  if(USE_APPS_SCRIPT)gsCall('updateRequestStatus',id,status).then(function(){return gsCall('getAllRequests');}).then(function(f){if(f)S.reqs=f;render();});
};
window.saveEdit=function(email){
  var d={email:email,name:document.getElementById('en').value,role:document.getElementById('er').value,startDate:document.getElementById('es').value,carryover:Number(document.getElementById('ec').value),managerEmail:document.getElementById('em').value,partTime:document.getElementById('ep').value==='yes'};
  S.emps=S.emps.map(function(e){return e.email===email?Object.assign({},e,d):e;});
  if(USE_APPS_SCRIPT)gsCall('updateEmployee',d);
  toast('Employee updated.','s');set({modal:null});
};
window.saveAdj=function(email){
  var delta=parseFloat(document.getElementById('ad').value);
  var note=document.getElementById('an').value;
  if(isNaN(delta)||delta===0){toast('Enter a non-zero amount.','w');return;}
  var a={id:Date.now().toString(),email:email,delta:delta,note:note,by:S.sess.email,at:new Date().toISOString()};
  S.adjs=[].concat(S.adjs,[a]);
  persist();
  if(USE_APPS_SCRIPT)gsCall('adjustBalance',email,delta,note).then(function(){return gsCall('getBalanceAdjustments');}).then(function(f){if(f)S.adjs=f;render();});
  toast('Balance adjusted.','s');set({modal:null});
};

function av(name,size){
  if(!name)return'';
  var ini=name.split(' ').map(function(p){return p[0];}).slice(0,2).join('').toUpperCase();
  var cols=[['#e8f4e6','#2d6a22'],['#eaf1fb','#1a4d7c'],['#fef8f0','#b5651d'],['#fdf0ef','#c0392b'],['#f0efe9','#5a5955']];
  var c=cols[name.charCodeAt(0)%cols.length];
  return'<span class="av" style="width:'+size+'px;height:'+size+'px;min-width:'+size+'px;background:'+c[0]+';color:'+c[1]+';font-size:'+Math.round(size*.36)+'px">'+ini+'</span>';
}
function bdg(text,cls){
  var map={Approved:'g',Denied:'r',Pending:'y',Cancelled:'gr',Admin:'g'};
  return'<span class="bdg '+(cls||map[text]||'gr')+'">'+text+'</span>';
}
