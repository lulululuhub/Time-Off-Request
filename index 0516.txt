function yrs(sd){const s=new Date(sd),n=new Date();let y=n.getFullYear()-s.getFullYear();if(n<new Date(n.getFullYear(),s.getMonth(),s.getDate()))y--;return Math.max(0,y);}
function allowance(e){
  // Part-time gets same allowance as full-time based on years of service
  return yrs(e.startDate)>=5?160:120;
}
function cap(e){return allowance(e)*1.5;}
function rate(e){return allowance(e)/24;}
function conseq(e){
  // Only difference for part-time: consecutive limit is 40h
  if(e.partTime)return 40;
  return yrs(e.startDate)>=5?120:80;
}
// Build a sorted list of all semi-monthly period dates in current year up to (not including) today
// Accrual posts the day AFTER the period closes: Apr 30 → available May 1
// Format a Date as YYYY-MM-DD using LOCAL time (not UTC)
// Avoids timezone shift where Apr 30 midnight local = Apr 29 UTC
function localDateStr(d){
  return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0');
}
function nextDayStr(dateStr){
  const d=new Date(dateStr+'T12:00:00');d.setDate(d.getDate()+1);return localDateStr(d);
}
// Count weekdays Mon-Fri inclusive between two date strings
function weekdaysInclusive(startStr,endStr){
  let count=0;
  const d=new Date(startStr+'T12:00:00'),end=new Date(endStr+'T12:00:00');
  while(d<=end){if(d.getDay()!==0&&d.getDay()!==6)count++;d.setDate(d.getDate()+1);}
  return count;
}
// Split a PTO request at semi-monthly period boundaries by weekday proportion
// e.g. 3/30-4/3 40h split at Mar31 → [{endDate:'2026-03-31',hours:16},{endDate:'2026-04-03',hours:24}]
function splitAtBoundaries(startStr,endStr,totalHours,periodStrs){
  const totalDays=weekdaysInclusive(startStr,endStr);
  if(totalDays===0)return [{endDate:endStr,hours:totalHours}];
  const hPerDay=totalHours/totalDays;
  const segments=[];
  let segStart=startStr,remaining=totalHours;
  for(let i=0;i<periodStrs.length;i++){
    const boundary=periodStrs[i];
    if(boundary>=segStart&&boundary<endStr){
      const daysInSeg=weekdaysInclusive(segStart,boundary);
      const hoursInSeg=Math.round(daysInSeg*hPerDay*100)/100;
      if(hoursInSeg>0){segments.push({endDate:boundary,hours:hoursInSeg});remaining=Math.round((remaining-hoursInSeg)*100)/100;}
      segStart=nextDayStr(boundary);
    }
  }
  if(remaining>0)segments.push({endDate:endStr,hours:remaining});
  return segments;
}
// Get completed semi-monthly period date strings in current year, strictly before today
function getPeriodDates(sd){
  const today=new Date(),todayStr=localDateStr(today);
  const hire=new Date(sd+'T12:00:00');
  const curYear=today.getFullYear();
  const effectiveStart=hire>new Date(curYear,0,1)?hire:new Date(curYear,0,1);
  const effectiveStr=localDateStr(effectiveStart);
  const dates=[];
  for(let mo=0;mo<12;mo++){
    const p15Str=localDateStr(new Date(curYear,mo,15));
    const peStr=localDateStr(new Date(curYear,mo+1,0));
    if(p15Str<todayStr&&p15Str>=effectiveStr)dates.push(p15Str);
    if(peStr<todayStr&&peStr>=effectiveStr)dates.push(peStr);
  }
  return dates.sort();
}
// Simulate balance period-by-period, splitting PTO requests at boundaries
// Correctly handles cap interactions when PTO spans a period boundary
function bal(e,reqs,adjs){
  const DEDUCTIBLE=['Paid Time Off'];
  const empRate=rate(e),empCap=cap(e);
  const empAdj=(adjs||[]).filter(a=>a.email===e.email).reduce((s,a)=>s+Number(a.delta),0);
  const periodStrs=getPeriodDates(e.startDate);
  const today=localDateStr(new Date());
  // Only deduct PTO whose START DATE has already passed or is today
  // Future approved PTO does not deduct until the leave actually begins
  const ptoSegments=[];
  (reqs||[]).filter(r=>r.employeeEmail===e.email&&r.status==='Approved'&&DEDUCTIBLE.includes(r.type)&&r.startDate<=today)
    .forEach(r=>splitAtBoundaries(r.startDate,r.endDate,Number(r.hours),periodStrs).forEach(seg=>ptoSegments.push(seg)));
  ptoSegments.sort((a,b)=>a.endDate.localeCompare(b.endDate));
  let balance=Math.min(Number(e.carryover||0)+empAdj,empCap),usedIdx=0;
  for(let i=0;i<periodStrs.length;i++){
    const pdStr=periodStrs[i];
    while(usedIdx<ptoSegments.length&&ptoSegments[usedIdx].endDate<=pdStr){
      balance=Math.max(0,balance-ptoSegments[usedIdx].hours);usedIdx++;
    }
    balance=Math.min(balance+empRate,empCap);
  }
  while(usedIdx<ptoSegments.length){balance=Math.max(0,balance-ptoSegments[usedIdx].hours);usedIdx++;}
  return Math.max(0,balance);
}
function accrued(e){
  const empRate=rate(e),empCap=cap(e);
  let balance=Math.min(Number(e.carryover||0),empCap);
  getPeriodDates(e.startDate).forEach(()=>{balance=Math.min(balance+empRate,empCap);});
  return balance;
}
function periods(sd){return getPeriodDates(sd).length;}
function todayStr(){return localDateStr(new Date());}
function pendHrs(e,reqs){return(reqs||[]).filter(r=>r.employeeEmail===e.email&&r.status==='Pending').reduce((s,r)=>s+Number(r.hours),0);}

function nextAccrual(){
  const t=new Date(),d=t.getDate(),m=t.getMonth(),y=t.getFullYear();
  const last=new Date(y,m+1,0).getDate();
  let n;if(d<15)n=new Date(y,m,15);else if(d<last)n=new Date(y,m+1,0);else n=new Date(y,m+1,15);
  return n.toISOString().slice(0,10);
}

// ── ACCRUAL SCHEDULE ──────────────────────────────────────────────
// Shows every future period with projected running balance
// Also used by the request form to show balance on a selected date
function getFutureAccrualSchedule(emp){
  const empRate=rate(emp);
  const empCap=cap(emp);
  const today=localDateStr(new Date());
  const curYear=new Date().getFullYear();
  const schedule=[];

  // Start from current balance
  let runningBal=bal(emp,S.reqs,S.adjs);

  // Get all future period dates in current year
  for(let mo=0;mo<12;mo++){
    const p15=localDateStr(new Date(curYear,mo,15));
    const pe=localDateStr(new Date(curYear,mo+1,0));
    if(p15>=today)schedule.push({date:p15,label:['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'][mo]+' 15'});
    if(pe>=today)schedule.push({date:pe,label:['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'][mo]+' '+new Date(curYear,mo+1,0).getDate()});
  }

  // Build projected balances — each period adds accrual (no PTO deductions, just showing potential)
  return schedule.map(p=>{
    runningBal=Math.min(runningBal+empRate,empCap);
    return {...p,balance:runningBal,rate:empRate};
  });
}

// Holiday warning on date change
window.checkHols=function(){
  const s=document.getElementById('rstart')?.value;
  const e=document.getElementById('rend')?.value;
  const holEl=document.getElementById('hol-warn');
  if(!holEl||!s||!e)return;

  // Holiday warning
  const hits=HOLIDAYS_2026.filter(h=>h.date>=s&&h.date<=e);
  if(hits.length>0){
    holEl.innerHTML=`<div class="alert info">Your date range includes ${hits.length} company holiday(s): ${hits.map(h=>`<strong>${h.name}</strong>`).join(', ')}. Holidays are paid and do not count against your PTO balance.</div>`;
  } else {
    holEl.innerHTML='';
  }

  // Show projected balance on start date
  const projEl=document.getElementById('proj-info');
  if(!projEl||!S.sess)return;
  const emp=S.emps.find(x=>x.email===S.sess.email);
  if(!emp)return;
  const today=localDateStr(new Date());
  const curBal=bal(emp,S.reqs,S.adjs);

  if(s<=today){
    // Date is today or past — show current balance
    projEl.innerHTML=`<div class="alert info" style="border-color:#b8ddb5;background:#eaf5e6;color:#2d6a22">
      <strong>Available balance: ${curBal.toFixed(2)}h</strong> as of today.
    </div>`;
  } else {
    // Future date — find projected balance from accrual schedule
    const schedule=getFutureAccrualSchedule(emp);
    // Find the last period BEFORE the start date
    let projBal=curBal;
    const periodsBeforeStart=schedule.filter(p=>p.date<s);
    if(periodsBeforeStart.length>0){
      projBal=periodsBeforeStart[periodsBeforeStart.length-1].balance;
    }
    const periodsCount=periodsBeforeStart.length;
    const startFmt=new Date(s+'T12:00:00').toLocaleDateString('en-US',{month:'long',day:'numeric',year:'numeric'});
    const nextPeriods=schedule.filter(p=>p.date>=today&&p.date<s);
    const periodList=nextPeriods.map(p=>`${p.label} (+${p.rate.toFixed(2)}h)`).join(', ');

    projEl.innerHTML=`<div style="background:#eaf5e6;border:2px solid #2d6a22;border-radius:10px;padding:14px 18px;margin:8px 0">
      <div style="display:flex;align-items:center;gap:10px;margin-bottom:${periodsCount>0?'8':'0'}px">
        <div style="width:36px;height:36px;border-radius:50%;background:#2d6a22;display:flex;align-items:center;justify-content:center;flex-shrink:0">
          <svg width="18" height="18" viewBox="0 0 16 16" fill="none" stroke="#fff" stroke-width="2"><circle cx="8" cy="8" r="6"/><path d="M8 5v3l2 2"/></svg>
        </div>
        <div>
          <div style="font-size:15px;font-weight:600;color:#1e5c1a">Projected balance on ${startFmt}</div>
          <div style="font-size:20px;font-weight:700;color:#2d6a22;letter-spacing:-0.5px">${projBal.toFixed(2)}h available</div>
        </div>
      </div>
      ${periodsCount>0?`<div style="font-size:12.5px;color:#5a8a55;background:#d5eed0;border-radius:6px;padding:7px 10px">
        Includes ${periodsCount} upcoming accrual${periodsCount>1?'s':''}: ${periodList}
      </div>`:''}
    </div>`;
  }
};

function empUsage(email){
  const approved=S.reqs.filter(r=>r.employeeEmail===email&&r.status==='Approved');
  const pto      =approved.filter(r=>r.type==='Paid Time Off').reduce((s,r)=>s+Number(r.hours),0);
  const unpaid   =approved.filter(r=>r.type==='Unpaid Time Off').reduce((s,r)=>s+Number(r.hours),0);
  const bereav   =approved.filter(r=>r.type==='Bereavement').reduce((s,r)=>s+Number(r.hours),0);
  const other    =approved.filter(r=>r.type==='Other').reduce((s,r)=>s+Number(r.hours),0);
  const pending  =S.reqs.filter(r=>r.employeeEmail===email&&r.status==='Pending').reduce((s,r)=>s+Number(r.hours),0);
  return {pto,unpaid,bereav,other,pending,total:pto+unpaid+bereav+other};
}

  S.reqs=S.reqs.map(r=>r.id===id?{...r,status:'Cancelled'}:r);
  set({etab:'history'});
  toast('Request cancelled.','w');
  if(USE_APPS_SCRIPT){

    }).then(function(freshReqs){

    });
  } else {

  }

window.doSubmit=async function(){
  const emp=S.emps.find(e=>e.email===S.sess.email);
  const type=document.getElementById('rtype').value;
  const start=document.getElementById('rstart').value;
  const end=document.getElementById('rend').value;
  const hours=parseFloat(document.getElementById('rhours').value)||0;
  const reason=document.getElementById('rreason').value;
  const errEl=document.getElementById('req-err');
  if(!start||!end){errEl.innerHTML=`<div class="alert err">Please select dates.</div>`;return;}
  if(new Date(start)>new Date(end)){errEl.innerHTML=`<div class="alert err">End date must be on or after start date.</div>`;return;}
  if(hours<4){errEl.innerHTML=`<div class="alert err">Minimum request is 4 hours.</div>`;return;}
  if(type==='Paid Time Off'){
    const b=bal(emp,S.reqs,S.adjs);
    if(hours>b){
      errEl.innerHTML=`<div class="alert err">Insufficient balance. You have ${b.toFixed(2)}h available today.</div>`;
      return;
    }
    if(hours>conseq(emp)){errEl.innerHTML=`<div class="alert err">Exceeds consecutive limit of ${conseq(emp)}h.</div>`;return;}
  }
  const req={id:Date.now().toString(),employeeEmail:S.sess.email,employeeName:S.sess.name,type,startDate:start,endDate:end,hours,reason,status:'Pending',createdAt:new Date().toISOString(),managerEmail:emp.managerEmail};
  if(USE_APPS_SCRIPT){
    // Add to local state immediately so it shows in history right away
    S.reqs=[...S.reqs,req];
    set({etab:'history'});
    toast('Request submitted! Your manager will be notified.','s');
    // Save to Sheet and refresh from Sheet to get the server-assigned ID
    gsCall('submitRequest',req).then(function(){
      return gsCall('getAllRequests');
    }).then(function(freshReqs){
      if(freshReqs)S.reqs=freshReqs;
      render();
    }).catch(function(e){
      console.error('Submit error:',e);
      toast('Saved locally but failed to sync. Please refresh.','w');
    });
  } else {
    S.reqs=[...S.reqs,req];
    persist();
    toast('Request submitted! Your manager will be notified.','s');
    set({etab:'history'});
  }
};

// ── AUTO-REFRESH ──────────────────────────────────────────────────
// Polls the Sheet every 30 seconds so changes made outside the app
// (e.g. manager clicks Approve/Deny in email) show up automatically.
var _refreshTimer=null;
function startAutoRefresh(){
  if(!USE_APPS_SCRIPT)return;
  stopAutoRefresh();
  _refreshTimer=setInterval(function(){
    if(!S.sess)return; // not logged in, skip
    gsCall('getAllRequests').then(function(freshReqs){
      if(!freshReqs)return;
      // Check if anything actually changed before re-rendering
      const oldJSON=JSON.stringify(S.reqs.map(r=>r.id+'_'+r.status).sort());
      const newJSON=JSON.stringify(freshReqs.map(r=>r.id+'_'+r.status).sort());
      if(oldJSON!==newJSON){
        S.reqs=freshReqs;
        render();
        // Show subtle notification if a pending request was approved/denied
        const changed=freshReqs.filter(function(nr){
          const old=S.reqs.find(function(or){return or.id===nr.id;});
          return old&&old.status==='Pending'&&nr.status!=='Pending';
        });
        if(changed.length>0){
          toast('A request status has been updated.','s');
        }
      }
    }).catch(function(e){
      console.log('Auto-refresh error (non-fatal):',e);
    });
  },30000); // every 30 seconds
}
function stopAutoRefresh(){
  if(_refreshTimer){clearInterval(_refreshTimer);_refreshTimer=null;}
}

function av(name,size){
  if(!name)return'';
  const ini=name.split(' ').map(p=>p[0]).slice(0,2).join('').toUpperCase();
  const cols=[['#e8f4e6','#2d6a22'],['#eaf1fb','#1a4d7c'],['#fef8f0','#b5651d'],['#fdf0ef','#c0392b'],['#f0efe9','#5a5955']];
  const c=cols[name.charCodeAt(0)%cols.length];
  return`<span class="av" style="width:${size}px;height:${size}px;min-width:${size}px;background:${c[0]};color:${c[1]};font-size:${Math.round(size*.36)}px">${ini}</span>`;
}
function bdg(text,cls){

  return`<span class="bdg ${cls||map[text]||'gr'}">${text}</span>`;
}
