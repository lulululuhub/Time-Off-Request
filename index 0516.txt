// ── WEB APP ENTRY POINT ──────────────────────────────────────────
function doGet(e) {
  var action    = e && e.parameter ? e.parameter.action    : null;
  var requestId = e && e.parameter ? e.parameter.requestId : null;
  var token     = e && e.parameter ? e.parameter.token     : null;

  // One-click approve/deny from manager email
  if ((action === 'approve' || action === 'deny') && requestId && token) {
    if (!validateToken(requestId, token)) {
      return HtmlService.createHtmlOutput('<h2 style="font-family:sans-serif">Invalid or expired link.</h2>');
    }
    var status = action === 'approve' ? 'Approved' : 'Denied';
    updateRequestStatus(requestId, status);
    return HtmlService.createHtmlOutput(
      '<div style="font-family:sans-serif;text-align:center;padding:60px 20px">'
      + '<h2 style="color:' + (action === 'approve' ? '#2d6a22' : '#c0392b') + '">Request ' + status + '</h2>'
      + '<p style="color:#7a7874">The employee has been notified automatically.</p>'
      + '<a href="' + ScriptApp.getService().getUrl() + '" style="color:#2d6a22">Go to PTO portal -></a>'
      + '</div>'
    );
  }

  // Normal app load
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('UIC PTO Tracker')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

// ── SESSION ───────────────────────────────────────────────────────
function getSession() {
  var email = Session.getActiveUser().getEmail();
  if (!email) return null;
  var emp = getEmployee(email);
  if (!emp) return null;
  return { email: emp.email, name: emp.name, role: emp.role };
}

// ── EMPLOYEES ────────────────────────────────────────────────────
function getEmployee(email) {
  var sheet = SS.getSheetByName('Employees');
  if (!sheet) return null;
  var rows    = sheet.getDataRange().getValues();
  var headers = rows[0];
  for (var i = 1; i < rows.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) {
      row[headers[j]] = rows[i][j];
    }
    if (row['email'] === email) {
      return cleanEmployee(row);
    }
  }
  return null;
}

function getAllEmployees() {
  var sheet = SS.getSheetByName('Employees');
  if (!sheet) return [];
  var rows    = sheet.getDataRange().getValues();
  var headers = rows[0];
  var result  = [];
  for (var i = 1; i < rows.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) {
      row[headers[j]] = rows[i][j];
    }
    if (row['email']) {
      result.push(cleanEmployee(row));
    }
  }
  return result;
}

// Normalize raw Sheet values into clean JS types
function cleanEmployee(row) {
  return {
    email:        String(row['email'] || '').trim(),
    name:         String(row['name']  || '').trim(),
    role:         String(row['role']  || 'Employee').trim(),
    startDate:    formatDate(row['startDate']),
    partTime:     toBoolean(row['partTime']),
    carryover:    toNumber(row['carryover']),
    managerEmail: String(row['managerEmail'] || '').trim()
  };
}

// ── REQUESTS ─────────────────────────────────────────────────────
function getAllRequests() {
  var sheet = SS.getSheetByName('Requests');
  if (!sheet || sheet.getLastRow() < 2) return [];
  var rows    = sheet.getDataRange().getValues();
  var headers = rows[0];
  var result  = [];
  for (var i = 1; i < rows.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) {
      row[headers[j]] = rows[i][j];
    }
    // Skip rows with no ID, no email, or zero hours (corrupt/empty rows)
    if (row['id'] && row['employeeEmail'] && parseFloat(row['hours']) > 0) {
      result.push({
        id:            String(row['id']),
        employeeEmail: String(row['employeeEmail'] || '').trim(),
        employeeName:  String(row['employeeName']  || '').trim(),
        type:          String(row['type']          || ''),
        startDate:     formatDate(row['startDate']),
        endDate:       formatDate(row['endDate']),
        hours:         toNumber(row['hours']),
        reason:        String(row['reason']        || ''),
        status:        String(row['status']        || 'Pending'),
        createdAt:     row['createdAt'] ? new Date(row['createdAt']).toISOString() : new Date().toISOString(),
        managerEmail:  String(row['managerEmail']  || '').trim()
      });
    }
  }
  return result;
}

function submitRequest(req) {
  var session = getSession();
  if (!session) throw new Error('Not authenticated');
  var sheet = SS.getSheetByName('Requests');
  var id    = Utilities.getUuid();
  var now   = new Date().toISOString();
  sheet.appendRow([
    id,
    session.email,
    session.name,
    req.type,
    req.startDate,
    req.endDate,
    req.hours,
    req.reason || '',
    'Pending',
    now,
    req.managerEmail || ''
  ]);
  req.id            = id;
  req.employeeEmail = session.email;
  req.employeeName  = session.name;
  req.status        = 'Pending';
  req.createdAt     = now;
  sendRequestNotification(req);
  return { success: true, id: id };
}

function updateRequestStatus(id, status) {
  var valid = ['Approved','Denied','Pending','Cancelled'];
  if (valid.indexOf(status) === -1) throw new Error('Invalid status: ' + status);
  var sheet = SS.getSheetByName('Requests');
  var rows  = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(id)) {
      sheet.getRange(i + 1, 9).setValue(status); // column 9 = status
      var headers = rows[0];
      var req = {};
      for (var j = 0; j < headers.length; j++) {
        req[headers[j]] = rows[i][j];
      }
      req['id']        = id;
      req['status']    = status;
      req['startDate'] = formatDate(req['startDate']);
      req['endDate']   = formatDate(req['endDate']);

      // Sync to Google Calendar
      if (status === 'Approved') {
        addPTOToCalendar(req);
      } else if (status === 'Cancelled' || status === 'Denied') {
        removePTOFromCalendar(id);
      }

      sendStatusNotification(req);
      logAudit('status_update', 'SYSTEM', id + ' -> ' + status);
      return { success: true };
    }
  }
  throw new Error('Request not found: ' + id);
}

// ── BALANCES / ADJUSTMENTS ───────────────────────────────────────
function getBalanceAdjustments() {
  var sheet = SS.getSheetByName('Balances');
  if (!sheet || sheet.getLastRow() < 2) return [];
  var rows    = sheet.getDataRange().getValues();
  var headers = rows[0];
  var result  = [];
  for (var i = 1; i < rows.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) {
      row[headers[j]] = rows[i][j];
    }
    // Only return MANUAL admin adjustments - skip SYSTEM accrual rows.
    // Accrual is calculated client-side based on hire date and periods.
    var by = String(row['by'] || '').trim();
    if (row['email'] && by !== 'SYSTEM') {
      result.push({
        id:    String(i),
        email: String(row['email'] || '').trim(),
        delta: toNumber(row['delta']),
        note:  String(row['note'] || ''),
        by:    by,
        at:    row['at'] ? new Date(row['at']).toISOString() : new Date().toISOString()
      });
    }
  }
  return result;
}

function adjustBalance(email, delta, note) {
  var session = getSession();
  if (!session || session.role !== 'Admin') throw new Error('Unauthorized');
  SS.getSheetByName('Balances').appendRow([
    email, new Date().toISOString(), delta, note, session.email
  ]);
  logAudit('balance_adjust', session.email, email + ': ' + delta + ' - ' + note);
  return { success: true };
}

// ── EMPLOYEE EDIT ────────────────────────────────────────────────
function updateEmployee(data) {
  var session = getSession();
  if (!session || session.role !== 'Admin') throw new Error('Unauthorized');
  var sheet   = SS.getSheetByName('Employees');
  var rows    = sheet.getDataRange().getValues();
  var headers = rows[0];
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]).trim() === String(data.email).trim()) {
      for (var j = 0; j < headers.length; j++) {
        var key = headers[j];
        if (data[key] !== undefined) {
          sheet.getRange(i + 1, j + 1).setValue(data[key]);
        }
      }
      logAudit('employee_update', session.email, 'Updated ' + data.email);
      return { success: true };
    }
  }
  throw new Error('Employee not found: ' + data.email);
}

}

// ── SEMI-MONTHLY ACCRUAL TRIGGER ─────────────────────────────────
// Run setupTrigger() once to install. Runs daily at 8am,
// but only processes on the 15th and last day of each month.
function runSemiMonthlyAccrual() {
  var today   = new Date();
  var day     = today.getDate();
  var lastDay = new Date(today.getFullYear(), today.getMonth() + 1, 0).getDate();
  // Accrual runs the day AFTER the period closes:
  // Period closes on 15th → run on 16th
  // Period closes on last day → run on 1st of next month
  var isAfter15th = (day === 16);
  var isFirstOfMonth = (day === 1);
  if (!isAfter15th && !isFirstOfMonth) return;

  var employees = getAllEmployees();
  var sheet     = SS.getSheetByName('Balances');
  employees.forEach(function(emp) {
    var yrs    = getYearsService(emp.startDate);
    // Part-time gets same allowance as full-time - only consecutive limit differs
    var annual = yrs >= 5 ? 160 : 120;
    var accrualPerPeriod = annual / 24;
    if (accrualPerPeriod > 0) {
      sheet.appendRow([
        emp.email,
        today.toISOString(),
        accrualPerPeriod,
        'Semi-monthly accrual',
        'SYSTEM'
      ]);
    }
  });
  logAudit('accrual_run', 'SYSTEM', 'Ran for ' + employees.length + ' employees on ' + today.toISOString());
}

function setupTrigger() {
  // Delete existing triggers first to avoid duplicates
  ScriptApp.getProjectTriggers().forEach(function(t) {
    if (t.getHandlerFunction() === 'runSemiMonthlyAccrual') {
      ScriptApp.deleteTrigger(t);
    }
  });
  ScriptApp.newTrigger('runSemiMonthlyAccrual')
    .timeBased()
    .everyDays(1)
    .atHour(8)
    .create();
  Logger.log('Trigger created: runSemiMonthlyAccrual runs daily at 8am');
}

// ── TOKEN SECURITY ────────────────────────────────────────────────
function makeToken(requestId) {
  var secret = PropertiesService.getScriptProperties().getProperty('TOKEN_SECRET');
  if (!secret) return 'no-secret';
  var bytes = Utilities.computeHmacSha256Signature(requestId, secret);
  return bytes.map(function(b) {
    return ('0' + (b & 0xFF).toString(16)).slice(-2);
  }).join('').slice(0, 16);
}

function validateToken(requestId, token) {
  return makeToken(requestId) === token;
}

function setupSecret() {
  PropertiesService.getScriptProperties().setProperty('TOKEN_SECRET', Utilities.getUuid());
  Logger.log('Secret created successfully');
}

// ── HELPERS ───────────────────────────────────────────────────────
// Convert Sheet date (Date object or string) to YYYY-MM-DD string
function formatDate(val) {
  if (!val) return '';
  if (val instanceof Date) {
    var y = val.getFullYear();
    var m = String(val.getMonth() + 1).padStart(2, '0');
    var d = String(val.getDate()).padStart(2, '0');
    return y + '-' + m + '-' + d;
  }
  // Already a string like "2015-04-16" or "2015-04-16T..."
  return String(val).slice(0, 10);
}

function toBoolean(val) {
  if (val === true || val === 'TRUE' || val === 'true' || val === 'Yes' || val === 'yes') return true;
  return false;
}

function toNumber(val) {
  var n = parseFloat(val);
  return isNaN(n) ? 0 : n;
}

function getYearsService(startDate) {
  var s = new Date(startDate);
  var n = new Date();
  var y = n.getFullYear() - s.getFullYear();
  if (n < new Date(n.getFullYear(), s.getMonth(), s.getDate())) y--;
  return Math.max(0, y);
}

// ── GOOGLE CALENDAR INTEGRATION ──────────────────────────────────
var COMPANY_CAL_ID = 'c_9afb084439b154efd07562acf496b2f05d3601a9e459cf6aaa09453470ab591d@group.calendar.google.com';

function getCompanyCal() {
  return CalendarApp.getCalendarById(COMPANY_CAL_ID);
}

// Called when a request is Approved - creates ALL-DAY event on company calendar
// All-day events show as color bars, not timed "9am" events
function addPTOToCalendar(req) {
  try {
    var cal = getCompanyCal();
    if (!cal) { Logger.log('Calendar not found: ' + COMPANY_CAL_ID); return; }

    var title = req.employeeName + '  ' + req.type;
    var desc  = 'Employee: ' + req.employeeName
              + ' | Type: '  + req.type
              + ' | Hours: ' + req.hours + 'h'
              + ' | Reason: ' + (req.reason || 'Not specified')
              + ' | Approved by PTO system';

    // Parse dates as local all-day dates
    var startParts = req.startDate.split('-');
    var endParts   = req.endDate.split('-');
    var startDate  = new Date(
      parseInt(startParts[0]),
      parseInt(startParts[1]) - 1,
      parseInt(startParts[2])
    );
    // All-day end date in Google Calendar is exclusive (add 1 day)
    var endDate = new Date(
      parseInt(endParts[0]),
      parseInt(endParts[1]) - 1,
      parseInt(endParts[2]) + 1
    );

    // Check if event already exists to avoid duplicates
    var checkEnd = new Date(endDate.getTime() - 1); // back to actual end day
    var existing = cal.getEvents(startDate, checkEnd);
    for (var i = 0; i < existing.length; i++) {
      if (existing[i].getTitle() === title) {
        Logger.log('Event already exists, skipping: ' + title);
        return;
      }
    }

    var event = cal.createAllDayEvent(title, startDate, endDate, {
      description: desc
    });
    event.setColor(CalendarApp.EventColor.GREEN); // Green for PTO
    storeCalEventId(req.id, event.getId());
    Logger.log('All-day calendar event created: ' + title);
  } catch(e) {
    Logger.log('Calendar error: ' + e.message);
  }
}

// Called when a request is Cancelled or Denied - removes event from calendar
function removePTOFromCalendar(requestId) {
  try {
    var eventId = getCalEventId(requestId);
    if (!eventId) return;
    var cal = getCompanyCal();
    if (!cal) return;
    var event = cal.getEventById(eventId);
    if (event) {
      event.deleteEvent();
      Logger.log('Calendar event deleted for request: ' + requestId);
    }
    clearCalEventId(requestId);
  } catch(e) {
    Logger.log('Calendar remove error: ' + e.message);
  }
}

// Add all holidays from the Holidays sheet to the company calendar
// Run this once manually each year after updating the Holidays tab
function syncHolidaysToCalendar() {
  var cal      = getCompanyCal();
  if (!cal) { Logger.log('Calendar not found!'); return; }
  var holidays = getHolidays();
  var added = 0, skipped = 0;

  holidays.forEach(function(h) {
    var date  = new Date(h.date + 'T12:00:00');
    var title = '[Holiday] ' + h.name + ' (UICU Holiday)';
    // Check if already exists
    var dayStart = new Date(h.date + 'T00:00:00');
    var dayEnd   = new Date(h.date + 'T23:59:59');
    var existing = cal.getEvents(dayStart, dayEnd);
    var found = false;
    for (var i = 0; i < existing.length; i++) {
      if (existing[i].getTitle() === title) { found = true; break; }
    }
    if (!found) {
      var hEvent = cal.createAllDayEvent(title, date, { description: 'UICU Paid Company Holiday' });
      hEvent.setColor(CalendarApp.EventColor.RED);
      added++;
    } else {
      skipped++;
    }
  });
  Logger.log('Holidays synced: ' + added + ' added, ' + skipped + ' already existed');
}

// ── ACCRUAL SHEETS GENERATOR ─────────────────────────────────────
// Run generateAccrualSheets() once to create a tab per employee
// showing their full 2026 accrual & PTO history like Fabian's spreadsheet.
// Re-run anytime to refresh all sheets with latest data.
function generateAccrualSheets() {
  var employees = getAllEmployees();
  var requests  = getAllRequests();
  var adjs      = getBalanceAdjustments();
  var ss        = SpreadsheetApp.getActiveSpreadsheet();
  var tz        = Session.getScriptTimeZone();
  var today     = Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd');
  var year      = new Date().getFullYear();

  var months    = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];

  employees.forEach(function(emp) {
    // ── Tab name ──────────────────────────────────────────────────
    var tabName = emp.name.replace(/\s+/g, '_').slice(0, 25) + '_Accrual';

    // Delete and recreate tab
    var existing = ss.getSheetByName(tabName);
    if (existing) ss.deleteSheet(existing);
    var sheet = ss.insertSheet(tabName);

    // ── Employee info header ──────────────────────────────────────
    sheet.getRange(1, 1, 1, 9).merge();
    sheet.getRange(1, 1).setValue('UICU PTO Accrual Statement 2026 - ' + emp.name);
    sheet.getRange(1, 1).setFontSize(13).setFontWeight('bold').setBackground('#2d6a22').setFontColor('#ffffff');

    // Info rows
    var yrs     = getYearsService(emp.startDate);
    var annual  = yrs >= 5 ? 160 : 120;
    var empRate = annual / 24;
    var empCap  = annual * 1.5;
    var carry   = parseFloat(emp.carryover) || 0;

    var infoData = [
      ['Email:', emp.email, '', 'Start Date:', emp.startDate, '', 'Years:', yrs, ''],
      ['Annual Allowance:', annual + 'h', '', 'Accrual Rate:', empRate.toFixed(4) + 'h/period', '', 'Cap:', empCap + 'h', ''],
      ['Carryover:', carry + 'h', '', 'Manager:', emp.managerEmail, '', 'Part-Time:', emp.partTime ? 'Yes' : 'No', ''],
    ];
    sheet.getRange(2, 1, 3, 9).setValues(infoData);
    sheet.getRange(2, 1, 3, 9).setBackground('#f5f4f1');
    sheet.getRange(2, 1).setFontWeight('bold');
    sheet.getRange(3, 1).setFontWeight('bold');
    sheet.getRange(4, 1).setFontWeight('bold');

    // Blank row
    sheet.getRange(5, 1, 1, 9).setBackground('#ffffff');

    // ── Column headers ────────────────────────────────────────────
    var headers = ['Period', 'Date', 'Opening Balance', 'Accrual (+)', 'PTO Deducted', 'Other Leave', 'Closing Balance', 'Leave Type', 'Request Dates'];
    sheet.getRange(6, 1, 1, 9).setValues([headers]);
    sheet.getRange(6, 1, 1, 9)
      .setBackground('#1a4d7c')
      .setFontColor('#ffffff')
      .setFontWeight('bold')
      .setHorizontalAlignment('center');

    // ── Build all period dates for the year ───────────────────────
    var periods = [];
    for (var mo = 0; mo < 12; mo++) {
      var p15Date  = new Date(year, mo, 15);
      var peDate   = new Date(year, mo + 1, 0);
      var p15Str   = Utilities.formatDate(p15Date, tz, 'yyyy-MM-dd');
      var peStr    = Utilities.formatDate(peDate,  tz, 'yyyy-MM-dd');
      var p15Label = months[mo] + ' 15';
      var peLabel  = months[mo] + ' ' + peDate.getDate();
      periods.push({ label: p15Label, date: p15Str });
      periods.push({ label: peLabel,  date: peStr  });
    }

    // ── Get this employee's approved PTO requests ─────────────────
    var empPTO = requests.filter(function(r) {
      return r.employeeEmail === emp.email && r.status === 'Approved';
    }).sort(function(a, b) { return a.startDate.localeCompare(b.startDate); });

    // Split a PTO request at period boundaries (mirrors Index.html logic)
    function weekdaysInclusive(startStr, endStr) {
      var count = 0;
      var d     = new Date(startStr + 'T12:00:00');
      var end   = new Date(endStr   + 'T12:00:00');
      while (d <= end) {
        var dow = d.getDay();
        if (dow !== 0 && dow !== 6) count++;
        d.setDate(d.getDate() + 1);
      }
      return count;
    }

    function nextDayStr(dateStr) {
      var d = new Date(dateStr + 'T12:00:00');
      d.setDate(d.getDate() + 1);
      return Utilities.formatDate(d, tz, 'yyyy-MM-dd');
    }

    function splitAtBoundaries(startStr, endStr, totalHours, periodStrs) {
      var totalDays = weekdaysInclusive(startStr, endStr);
      if (totalDays === 0) return [{ endDate: endStr, hours: totalHours, startDate: startStr }];
      var hPerDay   = totalHours / totalDays;
      var segments  = [];
      var segStart  = startStr;
      var remaining = totalHours;
      for (var i = 0; i < periodStrs.length; i++) {
        var boundary = periodStrs[i];
        if (boundary >= segStart && boundary < endStr) {
          var daysInSeg   = weekdaysInclusive(segStart, boundary);
          var hoursInSeg  = Math.round(daysInSeg * hPerDay * 100) / 100;
          if (hoursInSeg > 0) {
            segments.push({ endDate: boundary, hours: hoursInSeg, startDate: segStart });
            remaining = Math.round((remaining - hoursInSeg) * 100) / 100;
          }
          segStart = nextDayStr(boundary);
        }
      }
      if (remaining > 0) segments.push({ endDate: endStr, hours: remaining, startDate: segStart });
      return segments;
    }

    // Build all PTO segments split at period boundaries
    var periodStrs = periods.map(function(p) { return p.date; });
    var allSegments = [];
    empPTO.forEach(function(pto) {
      var segs = splitAtBoundaries(pto.startDate, pto.endDate, parseFloat(pto.hours), periodStrs);
      segs.forEach(function(seg) {
        allSegments.push({
          endDate:   seg.endDate,
          startDate: seg.startDate,
          hours:     seg.hours,
          type:      pto.type,
          reqDates:  pto.startDate + ' to ' + pto.endDate,
          origHours: pto.hours
        });
      });
    });
    allSegments.sort(function(a, b) { return a.endDate.localeCompare(b.endDate); });

    // Admin adjustments
    var empAdj = adjs.filter(function(a) { return a.email === emp.email; })
                     .reduce(function(s, a) { return s + parseFloat(a.delta); }, 0);

    // ── Simulate period by period ─────────────────────────────────
    var balance  = Math.min(carry + empAdj, empCap);
    var segIdx   = 0;
    var dataRows = [];

    // Opening carryover row
    dataRows.push([
      'Carryover',
      (year - 1) + '-12-31',
      '',
      '',
      '',
      '',
      balance.toFixed(2),
      '',
      'Prior year carryover: ' + carry + 'h' + (empAdj !== 0 ? ' | Adjustment: ' + empAdj + 'h' : '')
    ]);

    periods.forEach(function(period) {
      var pdStr      = period.date;
      var opening    = balance;
      var ptoDeduct  = 0;
      var otherLeave = 0;
      var leaveTypes = [];
      var reqDates   = [];

      // Deduct segments ending on or before this period (DEDUCTIBLE = Paid Time Off)
      while (segIdx < allSegments.length && allSegments[segIdx].endDate <= pdStr) {
        var seg = allSegments[segIdx];
        if (seg.type === 'Paid Time Off') {
          ptoDeduct += seg.hours;
        } else {
          otherLeave += seg.hours;
        }
        if (leaveTypes.indexOf(seg.type) === -1) leaveTypes.push(seg.type);
        if (reqDates.indexOf(seg.reqDates) === -1) reqDates.push(seg.reqDates);
        segIdx++;
      }

      // Deduct PTO from balance
      balance = Math.max(0, balance - ptoDeduct);

      // Always add accrual — for past periods it's actual, for future it's projected
      var isFuture = pdStr >= today;
      var beforeAccrual = balance;
      balance = Math.min(balance + empRate, empCap);
      var accrualAdded = balance - beforeAccrual;

      var closing = balance;

      dataRows.push([
        period.label,
        pdStr,
        opening.toFixed(2),
        accrualAdded.toFixed(2),
        ptoDeduct  > 0 ? ptoDeduct.toFixed(2)   : '',
        otherLeave > 0 ? otherLeave.toFixed(2)   : '',
        closing.toFixed(2),
        leaveTypes.length > 0 ? leaveTypes.join(', ') : (isFuture ? 'Projected' : ''),
        reqDates.join('; ')
      ]);
    });

    // Write data rows
    var startRow = 7;
    sheet.getRange(startRow, 1, dataRows.length, 9).setValues(dataRows);

    // ── Formatting ────────────────────────────────────────────────
    // Carryover row
    sheet.getRange(startRow, 1, 1, 9).setBackground('#e8f4e6').setFontWeight('bold');

    // Color period rows
    for (var r = 1; r < dataRows.length; r++) {
      var rowNum    = startRow + r;
      var rowData   = dataRows[r];
      var dateStr   = rowData[1];
      var hasPTO    = rowData[4] !== '';
      var hasOther  = rowData[5] !== '';
      var isFutRow  = dateStr >= today;
      var isAtCap   = parseFloat(String(rowData[6]).replace('~','')) >= empCap;

      if (isFutRow) {
        sheet.getRange(rowNum, 1, 1, 9).setBackground('#f0f8ff').setFontColor('#7a7874');
        sheet.getRange(rowNum, 7).setFontColor('#1a4d7c');
      } else if (hasPTO) {
        sheet.getRange(rowNum, 1, 1, 9).setBackground('#fff3e0');
        sheet.getRange(rowNum, 4).setFontColor('#2d6a22').setFontWeight('bold');
        sheet.getRange(rowNum, 5).setFontColor('#c0392b').setFontWeight('bold');
        sheet.getRange(rowNum, 7).setFontColor('#c06000').setFontWeight('bold');
      } else if (hasOther) {
        sheet.getRange(rowNum, 1, 1, 9).setBackground('#f5eaf8');
      } else if (r % 2 === 0) {
        sheet.getRange(rowNum, 1, 1, 9).setBackground('#f9f9f9');
      }

      if (isAtCap && !isFutRow) {
        sheet.getRange(rowNum, 7).setBackground('#fff3cd').setFontColor('#856404');
      }
    }

    // Summary row at bottom
    var summaryRow = startRow + dataRows.length;
    var totalPTO   = allSegments.filter(function(s) { return s.type === 'Paid Time Off'; })
                                .reduce(function(s, seg) { return s + seg.hours; }, 0);
    var totalOther = allSegments.filter(function(s) { return s.type !== 'Paid Time Off'; })
                                .reduce(function(s, seg) { return s + seg.hours; }, 0);

    sheet.getRange(summaryRow, 1, 1, 9).setValues([[
      'SUMMARY', '', 'Carryover: ' + carry + 'h',
      'Total accrued: ' + (periods.filter(function(p){return p.date<today;}).length * empRate).toFixed(2) + 'h',
      'PTO used: ' + totalPTO.toFixed(2) + 'h',
      'Other: ' + totalOther.toFixed(2) + 'h',
      'Current balance: ' + balance.toFixed(2) + 'h',
      '', ''
    ]]);
    sheet.getRange(summaryRow, 1, 1, 9)
      .setBackground('#2d6a22').setFontColor('#ffffff')
      .setFontWeight('bold');

    // Legend row
    var legendRow = summaryRow + 1;
    sheet.getRange(legendRow, 1, 1, 9).setValues([['LEGEND: White/Gray = Normal period | Orange = PTO used | Purple = Other leave | Blue = Future (projected) | Yellow = At cap', '', '', '', '', '', '', '', '']]);
    sheet.getRange(legendRow, 1, 1, 9).merge().setBackground('#f5f4f1').setFontColor('#7a7874').setFontSize(10);

    // ── Column widths & freeze ────────────────────────────────────
    sheet.setColumnWidth(1, 90);   // Period
    sheet.setColumnWidth(2, 100);  // Date
    sheet.setColumnWidth(3, 110);  // Opening
    sheet.setColumnWidth(4, 120);  // Accrual
    sheet.setColumnWidth(5, 120);  // PTO Deducted
    sheet.setColumnWidth(6, 100);  // Other Leave
    sheet.setColumnWidth(7, 120);  // Closing
    sheet.setColumnWidth(8, 140);  // Leave Type
    sheet.setColumnWidth(9, 200);  // Request Dates

    sheet.setFrozenRows(6);
    // Note: no column freeze — conflicts with merged header cell in row 1

    // Center-align numeric columns
    var totalDataRows = dataRows.length + 1;
    sheet.getRange(startRow, 2, totalDataRows, 6).setHorizontalAlignment('center');

    // Format numeric columns as numbers (2 decimal places) to prevent misinterpretation
    var numFormat = '0.00';
    sheet.getRange(startRow, 3, totalDataRows, 5).setNumberFormat(numFormat); // Opening, Accrual, PTO, Other, Closing
    // Opening balance col
    sheet.getRange(startRow, 3, totalDataRows, 1).setNumberFormat(numFormat);
    // Accrual col
    sheet.getRange(startRow, 4, totalDataRows, 1).setNumberFormat(numFormat);
    // PTO deducted col - positive number shown in red to indicate deduction
    sheet.getRange(startRow, 5, totalDataRows, 1).setNumberFormat('0.00');
    // Closing balance col
    sheet.getRange(startRow, 7, totalDataRows, 1).setNumberFormat(numFormat);

    Logger.log('Generated sheet for: ' + emp.name + ' (' + tabName + ')');
  });

  Logger.log('Done! Generated ' + employees.length + ' accrual sheets.');
  SpreadsheetApp.getUi().alert('Done! Generated accrual sheets for ' + employees.length + ' employees. Check the new tabs at the bottom of your spreadsheet.');
}

// Run this to manually sync an existing approved request to calendar
// Usage: set the requestId variable to the id from your Requests sheet
function manualSyncRequest() {
  var requestId = 'REPLACE_WITH_REQUEST_ID'; // e.g. '2026-001'
  var sheet = SS.getSheetByName('Requests');
  var rows  = sheet.getDataRange().getValues();
  var headers = rows[0];
  for (var i = 1; i < rows.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) row[headers[j]] = rows[i][j];
    if (String(row['id']) === requestId) {
      row['startDate'] = formatDate(row['startDate']);
      row['endDate']   = formatDate(row['endDate']);
      addPTOToCalendar(row);
      Logger.log('Synced request ' + requestId + ' to calendar');
      return;
    }
  }
  Logger.log('Request not found: ' + requestId);
}

// Sync ALL approved requests to calendar (run once to backfill)
function syncAllApprovedToCalendar() {
  var sheet = SS.getSheetByName('Requests');
  if (!sheet) { Logger.log('No Requests sheet!'); return; }
  var rows    = sheet.getDataRange().getValues();
  var headers = rows[0];
  var synced = 0, skipped = 0;
  for (var i = 1; i < rows.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) row[headers[j]] = rows[i][j];
    if (String(row['status']) === 'Approved') {
      row['startDate'] = formatDate(row['startDate']);
      row['endDate']   = formatDate(row['endDate']);
      addPTOToCalendar(row);
      synced++;
    } else {
      skipped++;
    }
  }
  Logger.log('Done! Synced: ' + synced + ' approved requests. Skipped: ' + skipped);
}

// ── CLEAN WIPE AND RESYNC ─────────────────────────────────────────
// Deletes ALL events from the calendar between two dates, then re-syncs
// from the Requests sheet. Use this to clean up duplicates.
// WARNING: This deletes ALL events in the date range including manual ones.
// Run wipeAndResyncCalendar() to execute.
function wipeCalendarEvents(fromDate, toDate) {
  var cal = getCompanyCal();
  if (!cal) { Logger.log('Calendar not found!'); return 0; }
  var events = cal.getEvents(new Date(fromDate), new Date(toDate));
  var deleted = 0;
  for (var i = 0; i < events.length; i++) {
    events[i].deleteEvent();
    deleted++;
  }
  Logger.log('Deleted ' + deleted + ' events between ' + fromDate + ' and ' + toDate);
  return deleted;
}

function wipeAndResyncCalendar() {
  // Step 1: Delete all events in 2026
  Logger.log('Step 1: Wiping all 2026 calendar events...');
  var deleted = wipeCalendarEvents('2026-01-01', '2026-12-31');
  Logger.log('Deleted ' + deleted + ' events.');

  // Step 2: Re-sync all approved PTO as clean all-day events
  Logger.log('Step 2: Re-syncing all approved requests as all-day events...');
  var sheet = SS.getSheetByName('Requests');
  if (!sheet) { Logger.log('No Requests sheet!'); return; }
  var rows    = sheet.getDataRange().getValues();
  var headers = rows[0];
  var synced  = 0;
  for (var i = 1; i < rows.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) row[headers[j]] = rows[i][j];
    if (String(row['status']) === 'Approved') {
      row['startDate'] = formatDate(row['startDate']);
      row['endDate']   = formatDate(row['endDate']);
      addPTOToCalendar(row);
      synced++;
    }
  }

  // Step 3: Re-sync holidays
  Logger.log('Step 3: Re-syncing holidays...');
  syncHolidaysToCalendar();

  Logger.log('Done! Wiped ' + deleted + ' old events, created ' + synced + ' clean all-day PTO events + holidays.');
}
